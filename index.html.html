Configurar Google Sheets

Conecte o app ao seu banco de dados

Passo a passo
1. Abra sua planilha no Google Sheets
2. Menu → Extensões → Apps Script
3. Cole o script abaixo, salve e clique em Implantar → Novo Implantação → App da Web. Acesso: Qualquer pessoa
4. Copie a URL gerada e cole no campo abaixo
Script para Google Apps Script
function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(e.parameter.sheet || 'Consultas'); var data = sheet.getDataRange().getValues(); var headers = data[0]; var rows = data.slice(1).map(function(row) { var obj = {}; headers.forEach(function(h, i) { obj[h] = row[i]; }); return obj; }); return ContentService .createTextOutput(JSON.stringify({data: rows})) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var body = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(body.sheet); sheet.appendRow(body.row); return ContentService .createTextOutput(JSON.stringify({ok: true})) .setMimeType(ContentService.MimeType.JSON); }
