Conectar Google Sheets

Configure a integração para salvar e sincronizar todos os dados da clínica no Google Sheets automaticamente.

  1. Acesse script.google.com e crie um novo projeto
  2. Cole o código abaixo no editor e salve (Ctrl+S)
  3. Clique em Implantar → Nova implantação → App da Web → Acesso: Qualquer pessoa
  4. Copie a URL gerada e cole no campo abaixo
Código Apps Script:
// ============================================ // Apps Script FINAL — cole e reimplante // ============================================ function doPost(e) { var result = processRequest(e.parameter, e.postData); return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var result = processRequest(e.parameter, null); // HtmlService permite CORS automático para GET return HtmlService.createHtmlOutput(JSON.stringify(result)); } function processRequest(params, postData) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = 'read'; var sheetName = ''; var rowData = null; // Ler parâmetros do GET if (params) { action = params.action || 'read'; sheetName = params.sheet || ''; rowData = params.row ? JSON.parse(params.row) : null; } // Sobrescrever com POST body se existir if (postData && postData.contents) { try { var body = JSON.parse(postData.contents); action = body.action || action; sheetName = body.sheet || sheetName; rowData = body.row || rowData; } catch(ex) {} } if (!sheetName) return { ok: false, error: 'sheet ausente' }; var sheet = ss.getSheetByName(sheetName); if (action === 'append') { if (!sheet) sheet = ss.insertSheet(sheetName); if (rowData) sheet.appendRow(rowData); return { ok: true, msg: 'Salvo' }; } if (action === 'read') { if (!sheet) return { ok: true, data: [] }; return { ok: true, data: sheet.getDataRange().getValues() }; } return { ok: false, error: 'acao invalida' }; } catch(err) { return { ok: false, error: err.message }; } }
